   14 -- Quick flag, ordered values, and a parameter matrix of permutation tuples.
   15 data OpApplySymmetries x = OpApplySymmetries Bool x x
   16   deriving (Eq, Ord, Show, Data, Functor, Traversable, Foldable, Typeable, Generic)
   17

   28 instance (TypeOf x, Pretty x, ExpressionLike x) => TypeOf (OpApplySymmetries x) where
   29   typeOf p@(OpApplySymmetries _ values symmetries) = do
   30     tv <- typeOf values
   31     ts <- typeOf symmetries
   32     case tv of
   33       TypeTuple _ -> return ()
   34       _ -> raiseTypeError $ "applySymmetries expects a tuple of values:" <+> pretty p
   35     entries <- case ts of
   36       TypeMatrix TypeInt{} (TypeTuple xs) -> return xs
   37       TypeList (TypeTuple xs) -> return xs
   38       _ -> raiseTypeError $ "applySymmetries expects a matrix of permutation tuples:" <+> pretty p
   39     domains <- forM entries $ \t -> case t of
   40       TypePermutation d -> return d
   41       _ -> raiseTypeError $ "applySymmetries entry is not a permutation:" <+> pretty p
   42     unless (length domains == length (nub domains)) $
   43       raiseTypeError $ "applySymmetries has multiple permutations for the same type:" <+> pretty p
   44     return TypeBool
   45

   49 instance Pretty x => Pretty (OpApplySymmetries x) where
